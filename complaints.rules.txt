// Add this match inside the existing /databases/{database}/documents block.
// Keep all existing rules for progress and notifications.
match /complaints/{id} {
  function signedIn() { return request.auth != null; }
  function admin() { return signedIn() && request.auth.token.email == 'gpeluso03@gmail.com' && request.auth.token.email_verified == true && request.auth.token.firebase.sign_in_provider == 'google.com'; }
  allow create: if signedIn()
    && request.resource.data.keys().hasOnly(['owner', 'title', 'text', 'contact', 'status', 'createdAt'])
    && request.resource.data.keys().hasAll(['owner', 'title', 'text', 'contact', 'status', 'createdAt'])
    && request.resource.data.owner == request.auth.uid
    && request.resource.data.title is string && request.resource.data.title.size() > 0 && request.resource.data.title.size() <= 120
    && request.resource.data.text is string && request.resource.data.text.size() > 0 && request.resource.data.text.size() <= 1500
    && request.resource.data.contact is string && request.resource.data.contact.size() <= 120
    && request.resource.data.status == 'new' && request.resource.data.createdAt == request.time;
  allow read: if signedIn() && (admin() || resource.data.owner == request.auth.uid);
  allow update: if admin() && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['status']) && request.resource.data.status in ['new', 'declined', 'resolved'];
  allow delete: if false;
}
